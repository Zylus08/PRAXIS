cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(Praxis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(praxis_core src/frontend/source.cpp src/frontend/lexer.cpp src/frontend/preprocessor.cpp src/frontend/parser.cpp src/ast/type.cpp src/ast/substitution.cpp src/semantics/classes.cpp src/semantics/overloads.cpp src/runtime/objects.cpp src/semantics/analyzer.cpp src/runtime/value.cpp src/runtime/interpreter.cpp src/native/bindings.cpp)
target_include_directories(praxis_core PUBLIC include)
if(MSVC)
 target_compile_options(praxis_core PUBLIC /W4 /permissive- /EHsc /utf-8 $<$<CONFIG:Debug>:/Od> $<$<CONFIG:Debug>:/Z7> $<$<CONFIG:Debug>:/RTC1> $<$<CONFIG:Release>:/O2>)
else()
 target_compile_options(praxis_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(praxis src/cli/main.cpp)
target_link_libraries(praxis PRIVATE praxis_core)
if(MSVC)
 target_link_options(praxis PRIVATE /STACK:16777216,4194304)
 target_link_libraries(praxis PRIVATE psapi)
endif()
enable_testing()
add_executable(praxis_unit tests/unit/frontend.cpp)
target_link_libraries(praxis_unit PRIVATE praxis_core)
add_test(NAME unit COMMAND praxis_unit)
add_executable(praxis_m2_unit tests/unit/m2.cpp)
target_link_libraries(praxis_m2_unit PRIVATE praxis_core)
add_test(NAME m2_unit COMMAND praxis_m2_unit)
add_executable(praxis_m3_unit tests/unit/m3.cpp)
target_link_libraries(praxis_m3_unit PRIVATE praxis_core)
add_test(NAME m3_unit COMMAND praxis_m3_unit)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME m3_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/m3/run.py $<TARGET_FILE:praxis>)
 add_test(NAME m2_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/m2/run.py $<TARGET_FILE:praxis>)
 add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/run.py $<TARGET_FILE:praxis>)
endif()
